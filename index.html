create or replace function public.handle_new_user() returns trigger language plpgsql security definer set search_path = public as $$ begin insert into public.members ( name, rank, role, active, auth_user_id ) values ( coalesce( new.raw_user_meta_data->>'name', split_part(new.email, '@', 1) ), 'Mitglied', 'Mitglied', true, new.id ); return new; end; $$; drop trigger if exists on_auth_user_created on auth.users; create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_user();
